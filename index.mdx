AuthKit Remix is the official authentication library for integrating [WorkOS & AuthKit](https://workos.com) into [Remix](https://remix.run) applications. It provides drop-in loader helpers, a callback route handler, and encrypted cookie-based session management — so you can add enterprise-grade authentication without building the plumbing yourself.

## Quick look

Add authentication to any Remix route loader with a single call:

```typescript app/routes/_index.tsx
import type { LoaderFunctionArgs } from '@remix-run/node';
import { useLoaderData } from '@remix-run/react';
import { authkitLoader } from '@workos-inc/authkit-remix';

export const loader = (args: LoaderFunctionArgs) => authkitLoader(args);

export default function HomePage() {
  const { user } = useLoaderData<typeof loader>();

  return (
    <div>
      {user ? (
        <p>Welcome back, {user.firstName}!</p>
      ) : (
        <a href="/sign-in">Sign in</a>
      )}
    </div>
  );
}
```
